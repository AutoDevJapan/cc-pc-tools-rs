func TestOther373TransportUnknownValuesPreserveQuotedTextAndProvenance(t *testing.T){
 raw:=[]byte(`{"unknown":[REDACTED],"literal":"[REDACTED]","escaped":"\"[REDACTED]","nested":[[REDACTED],{"x":[REDACTED]}]}`)
 normalized,p:=normalizeStreamTransport(raw)
 if p==nil||p.UnknownValues!=3||len(p.Offsets)!=3{t.Fatal("unknown transport values not accounted")}
 v,err:=streamJSON(normalized);if err!=nil{t.Fatal(err)}
 m:=v.(map[string]any)
 if m["unknown"]!=nil||m["literal"]!="[REDACTED]"||m["escaped"]!="\"[REDACTED]"{t.Fatal("hidden scalar inferred or quoted text changed")}
 canonical,err:=json.Marshal(v);if err!=nil{t.Fatal(err)}
 original:="OTHER373_CHUNK_V1\n"+string(raw)+"\n"
 r:=StreamReceipt{Original:original,OriginalSHA:streamHash([]byte(original)),Chunk:canonical,SHA:streamHash(canonical),Normalization:p}
 if err=verifyStreamNormalization(r);err!=nil{t.Fatal(err)}
 wire,err:=json.MarshalIndent(r,"","  ");if err!=nil{t.Fatal(err)}
 var roundtrip StreamReceipt;if err=json.Unmarshal(wire,&roundtrip);err!=nil{t.Fatal(err)}
 if err=verifyStreamNormalization(roundtrip);err!=nil{t.Fatalf("saved indented receipt must retain canonical meaning: %v",err)}
 r.Normalization.Offsets[0]++
 if verifyStreamNormalization(r)==nil{t.Fatal("forged transform offset accepted")}
 for _,invalid:=range []string{`{"x":[REDACTED]true}`,`{"x":true,"x":[REDACTED]}`,`{"x":[OTHER]}`} {
  b,_:=normalizeStreamTransport([]byte(invalid));if _,e:=streamJSON(b);e==nil{t.Fatal("non-transport syntax/duplicate accepted")}
 }
}
