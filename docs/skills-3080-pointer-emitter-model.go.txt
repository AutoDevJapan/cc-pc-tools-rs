package emittermath

import (
	"fmt"
	"math"
	"testing"
)

// Numerical acceptance model, authored here for review. This does not run an
// upstream helper, render a product, or claim a browser implementation exists.
type point struct{ x, y float64 }
type emitter struct {
	last  point
	carry float64
	ready bool
}

func (e *emitter) reset() { *e = emitter{} }

func (e *emitter) advance(p point, step float64, cap int) ([]point, float64, error) {
	finite := func(x float64) bool { return !math.IsNaN(x) && !math.IsInf(x, 0) }
	if !finite(p.x) || !finite(p.y) || !finite(step) || step <= 0 || cap < 1 {
		return nil, 0, fmt.Errorf("finite coordinates, positive step and cap required")
	}
	if !e.ready {
		e.last, e.ready = p, true
		return nil, 0, nil
	}
	dx, dy := p.x-e.last.x, p.y-e.last.y
	d := math.Hypot(dx, dy)
	if d == 0 {
		return nil, 0, nil
	}
	total := e.carry + d
	if !finite(total) {
		return nil, 0, fmt.Errorf("distance overflow")
	}
	wanted := math.Floor(total / step)
	count := int(math.Min(wanted, float64(cap)))
	points := make([]point, 0, count)
	for k := 1; k <= count; k++ {
		fraction := (float64(k)*step - e.carry) / d
		fraction = math.Max(0, math.Min(1, fraction))
		points = append(points, point{e.last.x + dx*fraction, e.last.y + dy*fraction})
	}
	// Deliberately drop complete overdue steps after a cap; never replay them
	// from a stationary position during later frames.
	e.carry, e.last = math.Mod(total, step), p
	return points, wanted - float64(count), nil
}

func sampled(t *testing.T, spacing float64) []point {
	t.Helper()
	e := emitter{}
	_, _, _ = e.advance(point{}, 3, 100)
	var result []point
	for x := spacing; x <= 100; x += spacing {
		got, dropped, err := e.advance(point{x, 0}, 3, 100)
		if err != nil || dropped != 0 || e.carry < 0 || e.carry >= 3 {
			t.Fatalf("invalid uncapped advance: %v %g", err, dropped)
		}
		result = append(result, got...)
	}
	return result
}

func TestSpacingAndCountDoNotDependOnStraightPathSampling(t *testing.T) {
	for _, sample := range []float64{1, 2, 4, 10, 20, 100} {
		points := sampled(t, sample)
		if len(points) != 33 {
			t.Fatalf("sample %g: count %d", sample, len(points))
		}
		for i, p := range points {
			if math.Abs(p.x-float64((i+1)*3)) > 1e-10 || p.y != 0 {
				t.Fatalf("sample %g: point %d = %v", sample, i, p)
			}
		}
	}
}

func TestFormerPlacementFormulaFailsTheSmallSegmentCase(t *testing.T) {
	// Transcription of only the arithmetic from the public text, using harmless
	// synthetic positions. No downloaded JavaScript is executed.
	carry := 0.0
	var previous float64
	foundUnequalGap := false
	for x := 2.0; x <= 100; x += 2 {
		carry += 2
		guard := 0
		for carry >= 3 && guard < 14 {
			guard++
			carry -= 3
			p := x - 2 + 2*math.Min(1, float64(guard)*3/2)
			if previous != 0 && math.Abs(p-previous-3) > 1e-10 {
				foundUnequalGap = true
			}
			previous = p
		}
	}
	if !foundUnequalGap {
		t.Fatal("regression fixture did not distinguish the former formula")
	}
}

func TestTeleportCapDropsBacklogInsteadOfEmittingItAtRest(t *testing.T) {
	e := emitter{}
	_, _, _ = e.advance(point{}, 3, 14)
	points, dropped, err := e.advance(point{1000, 0}, 3, 14)
	if err != nil || len(points) != 14 || dropped != 319 || e.carry != 1 {
		t.Fatalf("teleport: %v %d %g carry%g", err, len(points), dropped, e.carry)
	}
	// This model deliberately keeps the first capped samples in the interval,
	// not the final samples near the cursor or a uniformly thinned subset.
	for i, p := range points {
		if p.x != float64((i+1)*3) || p.y != 0 {
			t.Fatal("cap did not retain the first interval samples")
		}
	}
	for i := 0; i < 240; i++ {
		points, dropped, err = e.advance(point{1000, 0}, 3, 14)
		if err != nil || len(points) != 0 || dropped != 0 {
			t.Fatal("stationary pointer replayed old distance")
		}
	}
	points, dropped, err = e.advance(point{1002, 0}, 3, 14)
	if err != nil || dropped != 0 || len(points) != 1 || points[0].x != 1002 {
		t.Fatal("cap lost fractional remainder")
	}
}

func TestDiagonalSpacingAndReset(t *testing.T) {
	e := emitter{}
	_, _, _ = e.advance(point{}, 5, 14)
	points, dropped, err := e.advance(point{12, 16}, 5, 14)
	if err != nil || dropped != 0 || len(points) != 4 {
		t.Fatal("diagonal count differs")
	}
	for i, p := range points {
		if math.Abs(math.Hypot(p.x, p.y)-float64((i+1)*5)) > 1e-10 {
			t.Fatal("diagonal spacing differs")
		}
	}
	e.reset()
	points, _, err = e.advance(point{900, 900}, 5, 14)
	if err != nil || len(points) != 0 || e.carry != 0 {
		t.Fatal("reset bridged an inactive interval")
	}
}

func TestInvalidInputLeavesStateUnchanged(t *testing.T) {
	e := emitter{last: point{2, 3}, carry: 1, ready: true}
	before := e
	for _, bad := range []struct {
		p    point
		step float64
		cap  int
	}{{point{math.NaN(), 0}, 3, 14}, {point{}, 0, 14}, {point{}, 3, 0}} {
		if _, _, err := e.advance(bad.p, bad.step, bad.cap); err == nil || e != before {
			t.Fatal("invalid input was accepted or mutated state")
		}
	}
}
