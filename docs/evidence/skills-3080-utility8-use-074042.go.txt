func u8Need(t *testing.T, ok bool, why string) {
	t.Helper()
	if !ok {
		t.Fatal(why)

func u8SHA(b []byte) string { v := sha256.Sum256(b); return hex.EncodeToString(v[:]) }
func u8Read(p string) ([]byte, error) {
	i, e := os.Lstat(p)
	if e != nil {
		return nil, e
	}
	if !i.Mode().IsRegular() || i.Size() > 2<<20 {
		return nil, errors.New("bounded regular file required")
	}
	w, e := syscall.UTF16PtrFromString(p)
	if e != nil {
		return nil, e
	}
	h, e := syscall.CreateFile(w, syscall.GENERIC_READ, syscall.FILE_SHARE_READ|syscall.FILE_SHARE_WRITE|syscall.FILE_SHARE_DELETE, nil, syscall.OPEN_EXISTING, syscall.FILE_FLAG_OPEN_REPARSE_POINT, 0)
	if e != nil {
		return nil, e
	}
	f := os.NewFile(uintptr(h), p)
	defer f.Close()
	st, e := f.Stat()
	var info syscall.ByHandleFileInformation
	if e != nil || !os.SameFile(i, st) || syscall.GetFileInformationByHandle(h, &info) != nil || info.FileAttributes&syscall.FILE_ATTRIBUTE_REPARSE_POINT != 0 || info.NumberOfLinks != 1 {
		return nil, errors.New("identity/reparse/link boundary differs")
	}
	b, e := io.ReadAll(io.LimitReader(f, 2<<20+1))
	if e != nil || len(b) > 2<<20 {
		return nil, errors.New("bounded read failed")
	}
	return b, nil
}
func u8Save(t *testing.T, name string, v any) {
	t.Helper()
	b, e := json.MarshalIndent(v, "", "  ")
	if e != nil {
		t.Fatal(e)
	}
	f, e := os.OpenFile(filepath.Join(utility8P, name), os.O_CREATE|os.O_EXCL|os.O_WRONLY, 0600)
	if e != nil {
		t.Fatal(e)
	}
	_, w := f.Write(append(b, '\n'))
	s := f.Sync()
	c := f.Close()
	if e = errors.Join(w, s, c); e != nil {
		t.Fatal(e)
	}
}

	if e != nil {
		t.Fatal(e)

		Files []struct {
			Target  string
			SHA     string `json:"sha256"`
			Content string `json:"content_base64"`

		Sources []struct {
			Path string
			SHA  string `json:"sha256"`
		}
	}
	u8Need(t, json.Unmarshal(manifest, &m) == nil && len(m.Files) == 48 && len(m.Sources) == 8, "manifest scope differs")
	allowed := map[string]bool{"build-game-changelog": true, "test-playable-web-games": true, "audit-reference-originality": true, "browser-video-recording": true, "stitched-full-page-capture": true, "video-to-superprompt": true, "html-to-interaction-prompts": true, "optimize-web-animations": true}
	out := u8Installed{map[string][]byte{}, map[string]string{}, map[string]string{}, created.Commit}

	for _, s := range m.Sources {
		parts := strings.Split(s.Path, "/")

		out.Sources[parts[len(parts)-2]] = s.SHA
	}
	for _, r := range m.Files {
		parts := strings.SplitN(r.Target, "/", 2)
		u8Need(t, len(parts) == 2 && allowed[parts[0]] && !strings.ContainsAny(r.Target, "\\:") && out.Files[r.Target] == nil, "unexpected/duplicate target")

			if strings.EqualFold(dir, root) {
				break
			}
			u8Need(t, filepath.Dir(dir) != dir, "root not reached")
		}

			t.Fatal(e)
		}

type u8Deployment struct{ ID, Version, Source, Date, State string }

func u8Ledger(in []u8Deployment) ([]u8Deployment, error) {
	versions, ids := map[string]bool{}, map[string]bool{}
	out := []u8Deployment{}
	for _, r := range in {
		if r.State != "succeeded" {
			continue
		}
		if r.ID == "" || r.Version == "" || ids[r.ID] || versions[r.Version] {
			return nil, errors.New("duplicate/missing release identity")
		}
		if r.Date != "" {
			if _, e := time.Parse("2006-01-02", r.Date); e != nil {
				return nil, e
			}
		}
		if r.Source != "" && !regexp.MustCompile(`^[0-9a-f]{40}$`).MatchString(r.Source) {
			return nil, errors.New("invalid source identity")
		}
		ids[r.ID] = true
		versions[r.Version] = true
		out = append([]u8Deployment{r}, out...)
	}
	return out, nil
}
func u8Release(t *testing.T) any {
	a, c := strings.Repeat("a", 40), strings.Repeat("c", 40)
	rows := []u8Deployment{{"d100", "2.5.1", a, "2026-10-01", "succeeded"}, {"d101", "2.5.2", c, "2026-10-02", "failed"}, {"d102", "2.5.2", c, "2026-10-03", "succeeded"}, {"d103", "2.5.3", a, "2026-10-04", "succeeded"}, {"d104", "2.5.4", "", "", "draft"}}
	ledger, e := u8Ledger(rows)
	u8Need(t, e == nil && len(ledger) == 3 && ledger[0].Version == "2.5.3" && ledger[0].Source == ledger[2].Source, "rollback/failed/draft release mapping differs")
	_, e = u8Ledger(append(rows, u8Deployment{"d105", "2.5.3", a, "2026-10-05", "succeeded"}))
	u8Need(t, e != nil, "duplicate version accepted")
	_, e = u8Ledger([]u8Deployment{{"bad", "existing-v", a, "2026-02-30", "succeeded"}})
	u8Need(t, e != nil, "impossible date accepted")
	unknown, e := u8Ledger([]u8Deployment{{"unknown", "existing-v", "", "", "succeeded"}})
	u8Need(t, e == nil && unknown[0].Source == "" && unknown[0].Date == "", "unknown provenance invented")
	return map[string]any{"fixture_ledger": ledger, "latest_shipped_version": "2.5.3", "draft_not_shipped": true, "rollback_same_source_retained": true, "unknown_provenance_kept": true, "negative_duplicate_and_invalid_date_rejected": true, "real_deployment_or_menu_UI": false}
}

type u8Evidence struct {
	Journey, Kind string
	Passed        bool
}

func u8Satisfied(journey, required string, observations []u8Evidence) bool {
	for _, v := range observations {
		if v.Journey == journey && v.Kind == required && v.Passed {
			return true
		}
	}
	return false
}
func u8QA(t *testing.T) any {
	obs := []u8Evidence{{"save-migration", "deterministic-fixture", true}, {"render-feedback", "mock", true}}
	u8Need(t, u8Satisfied("save-migration", "deterministic-fixture", obs), "actual fixture evidence not accepted for its declared kind")
	u8Need(t, !u8Satisfied("save-migration", "real-browser", obs) && !u8Satisfied("render-feedback", "real-browser", obs), "fixture/mock promoted to real play")
	return map[string]any{"fixture_scope_satisfied": true, "real_browser_scope_satisfied": false, "unperformed_real_play_explicit": true, "selected_journeys": []string{"save-migration", "render-feedback"}, "no_nonexistent_boss_or_mobile_requirement_added": true}
}
func u8Originality(t *testing.T, in u8Installed) any {
	rows := []map[string]any{}
	for name, source := range in.Sources {
		original := in.Files[name+"/references/upstream-original.txt"]
		notice := in.Files[name+"/references/LICENSE.txt"]
		active := in.Files[name+"/SKILL.md"]
		u8Need(t, u8SHA(original) == source && u8SHA(notice) == "b2a071ba22fefd803f4e9d1f5c777dd79c62d191695586a1a73fd4d0ebd051c6", "original/notice mismatch")
		u8Need(t, !bytes.Equal(original, active), "active entry was copied verbatim")
		rows = append(rows, map[string]any{"target": name, "current_evidence": name + "/references/upstream-original.txt", "reference_evidence": "fixed manifest source SHA256 " + source, "exact_bytes": true, "role": "retained upstream procedure for audit", "exact_author_notice_retained": true, "active_is_not_identical": true, "legal_conclusion": false})
	}
	classify := func(exact, noticeKnown, published bool) string {
		if exact && published && !noticeKnown {
			return "conditions_unknown_requires_review"
		}
		if exact && noticeKnown {
			return "recorded_exact_copy_with_notice"
		}
		return "no_exact_match_in_checked_bytes"
	}
	u8Need(t, classify(true, true, false) == "recorded_exact_copy_with_notice" && classify(true, false, true) != "clear", "exact match/permission/context conflated")
	return map[string]any{"actual_installed_archive_comparisons": rows, "matching_bytes_not_automatic_violation": true, "unknown_conditions_not_passed": true, "legal_verdict_or_automatic_deletion": false, "scope": "Only the eight newly created packages; no unknown central bodies or supplied third-party media."}
}

type u8Shot struct {
	Click, Result float64
	X, Y          float64
}

func u8ShotPoint(s u8Shot) ([2]float64, error) {
	if s.Result < s.Click || s.X < 50 || s.X >= 250 || s.Y < 20 || s.Y >= 170 {
		return [2]float64{}, errors.New("timeline/crop mismatch")
	}
	return [2]float64{(s.X - 50) * 2, (s.Y - 20) * 2}, nil
}
func u8Recording(t *testing.T) any {
	p, e := u8ShotPoint(u8Shot{.5, .75, 150, 80})
	u8Need(t, e == nil && p == [2]float64{200, 120}, "source/crop/output coordinates differ")
	_, e = u8ShotPoint(u8Shot{.5, .25, 150, 80})
	u8Need(t, e != nil, "result before click accepted")
	_, e = u8ShotPoint(u8Shot{.5, .75, 500, 80})
	u8Need(t, e != nil, "out-of-crop cursor silently clamped")
	return map[string]any{"data_model_output_point": p, "cursor_top_left_for_output_pixel_hotspot_2_3": [2]float64{198, 117}, "result_before_click_and_outside_crop_rejected": true, "actual_cursor_asset_or_browser_recording": false, "synthetic_video_is_not_continuous_browser_recording": true}

type u8Segment struct{ From, Through, Width int }

func u8Coverage(parts []u8Segment, height, width int) bool {
	next := 0
	for _, p := range parts {
		if p.From != next || p.Through <= p.From || p.Width != width {
			return false
		}
		next = p.Through
	}
	return next == height
}
func u8Stitch(t *testing.T) any {
	good := []u8Segment{{0, 600, 1280}, {600, 1200, 1280}, {1200, 1440, 1280}}
	u8Need(t, u8Coverage(good, 1440, 1280), "contiguous half-open coverage rejected")
	gap := append([]u8Segment{}, good...)
	gap[1].From = 601
	u8Need(t, !u8Coverage(gap, 1440, 1280), "missing row passed")
	width := append([]u8Segment{}, good...)
	width[1].Width = 1440
	u8Need(t, !u8Coverage(width, 1440, 1280), "mixed viewport passed")
	return map[string]any{"declared_document_ranges": good, "range_coverage_valid": true, "gap_and_mixed_width_rejected": true, "image_content_sticky_header_DPR_or_real_page_proved": false}
}

type u8Animation struct {
	Visible, Disposed, Frame                   bool
	Generation, SimulationTicks, DiscardedLate int
	SharedAlive                                bool
}

func (s *u8Animation) visibility(v bool) { s.Visible = v; s.Frame = v && !s.Disposed }
func (s *u8Animation) dispose()          { s.Disposed = true; s.Generation++; s.Frame = false }
func (s *u8Animation) load(g int) {
	if s.Disposed || g != s.Generation {
		s.DiscardedLate++
		return
	}
	s.Frame = s.Visible
}
func u8Animations(t *testing.T) any {
	s := u8Animation{Generation: 1, SharedAlive: true}
	s.visibility(true)
	s.visibility(true)
	u8Need(t, s.Frame, "visible loop did not start")
	s.visibility(false)
	s.SimulationTicks++
	u8Need(t, !s.Frame && s.SimulationTicks == 1, "decorative stop changed required simulation")
	s.visibility(true)
	u8Need(t, s.Frame, "re-entry not resumed")
	s.dispose()
	s.load(1)
	u8Need(t, !s.Frame && s.DiscardedLate == 1 && s.SharedAlive, "late result resurrected disposed owner or destroyed another owner resource")
	return map[string]any{"bounded_state_model": s, "negative_late_loader_rejected": true, "shared_other_owner_resource_preserved": true, "real_CSS_RAF_or_GPU_profile": false, "heap_leak_ruled_out": false}
}
