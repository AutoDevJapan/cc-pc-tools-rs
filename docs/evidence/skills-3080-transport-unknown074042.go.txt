// The transport may replace an entire JSON scalar by the non-JSON sentinel
// [REDACTED]. Preserve that value as null, never infer the hidden scalar. Quoted
// text and every other byte stay unchanged. Strict parsing/duplicate checks and
// all mandatory source/reading/boundary checks still run afterwards.
func normalizeStreamTransport(raw []byte) ([]byte, *StreamTransportNormalization) {
	const sentinel="[REDACTED]"
	var out []byte
	var offsets []int
	inString, escaped:=false,false
	last:=0
	for i:=0;i<len(raw);i++ {
		c:=raw[i]
		if inString {
			if escaped { escaped=false;continue }
			if c=='\\' { escaped=true;continue }
			if c=='"' { inString=false }
			continue
		}
		if c=='"' { inString=true;continue }
		if !bytes.HasPrefix(raw[i:],[]byte(sentinel)){continue}
		before:=i-1;for before>=0 && (raw[before]==' '||raw[before]=='\t'||raw[before]=='\n'||raw[before]=='\r'){before--}
		after:=i+len(sentinel);for after<len(raw)&&(raw[after]==' '||raw[after]=='\t'||raw[after]=='\n'||raw[after]=='\r'){after++}
		if before<0 || !strings.ContainsRune(":[,",rune(raw[before])) || (after<len(raw)&&!strings.ContainsRune(",]}",rune(raw[after]))){continue}
		out=append(out,raw[last:i]...);out=append(out,"null"...)
		offsets=append(offsets,i);i+=len(sentinel)-1;last=i+1
	}
	if len(offsets)==0{return raw,nil}
	out=append(out,raw[last:]...)
	return out,&StreamTransportNormalization{Domain:"transport-redacted-value-to-unknown-null-v1",InputSHA:streamHash(raw),NormalizedSHA:streamHash(out),Offsets:offsets,UnknownValues:len(offsets)}
}

func verifyStreamNormalization(r StreamReceipt) error {
	if r.Normalization==nil{return nil}
	original:=[]byte(r.Original)
	if streamHash(original)!=r.OriginalSHA{return errors.New("normalization original differs")}
	begin,end:=bytes.IndexByte(original,'{'),bytes.LastIndexByte(original,'}')
	if begin<0||end<=begin{return errors.New("normalization origin envelope missing")}
	normalized,proof:=normalizeStreamTransport(original[begin:end+1])
	want,_:=json.Marshal(proof);got,_:=json.Marshal(r.Normalization)
	if proof==nil||!bytes.Equal(want,got){return errors.New("normalization provenance differs")}
	v,e:=streamJSON(normalized);if e!=nil{return e};canonical,e:=json.Marshal(v);if e!=nil{return e}
	storedValue,e:=streamJSON(r.Chunk);if e!=nil{return e};storedCanonical,e:=json.Marshal(storedValue);if e!=nil{return e}
	if !bytes.Equal(canonical,storedCanonical)||streamHash(canonical)!=r.SHA{return errors.New("normalization changed canonical meaning")}
	return nil
}

func streamHash(b []byte) string { h := sha256.Sum256(b); return hex.EncodeToString(h[:]) }
func streamRead(path string) ([]byte, error) {
	s, e := os.Lstat(path)
	if e != nil {
		return nil, e
	}
	if !s.Mode().IsRegular() || s.Size() > 2<<20 {
		return nil, errors.New("bounded regular metadata required")
	}
	wide, e := syscall.UTF16PtrFromString(path)
	if e != nil {
		return nil, e
	}
	handle, e := syscall.CreateFile(wide, syscall.GENERIC_READ, syscall.FILE_SHARE_READ|syscall.FILE_SHARE_WRITE|syscall.FILE_SHARE_DELETE, nil, syscall.OPEN_EXISTING, syscall.FILE_ATTRIBUTE_NORMAL|syscall.FILE_FLAG_OPEN_REPARSE_POINT, 0)
	if e != nil {
		return nil, e
	}
	f := os.NewFile(uintptr(handle), path)
	if f == nil {
		syscall.CloseHandle(handle)
		return nil, errors.New("metadata handle unavailable")
	}
	defer f.Close()
	opened, e := f.Stat()
	if e != nil || !opened.Mode().IsRegular() || !os.SameFile(s, opened) {
		return nil, errors.New("metadata changed during open; retry")
	}
	b, e := io.ReadAll(io.LimitReader(f, (2<<20)+1))
	if e != nil || len(b) > 2<<20 {
		return nil, errors.New("metadata read failed or exceeded bound")
	}
	return b, nil
}
func streamNew(path string, b []byte) error {
	if _, e := os.Lstat(path); e == nil {
		old, re := streamRead(path)
		if re == nil && bytes.Equal(old, b) {
			return nil
		}
		return errors.New("different existing receipt retained")
	}
	f, e := os.CreateTemp(filepath.Dir(path), ".other373-receipt-*")
	if e != nil {
		return e
	}
	defer os.Remove(f.Name())
	_, we := f.Write(b)
	se := f.Sync()
	ce := f.Close()
	if e := errors.Join(we, se, ce); e != nil {
		return e
	}
	return os.Link(f.Name(), path)
}
func streamEncode(v any) ([]byte, error) {
	b, e := json.MarshalIndent(v, "", "  ")
	return append(b, '\n'), e
}
func streamJSON(raw []byte) (any, error) {
	if !utf8.Valid(raw) {
		return nil, errors.New("non-UTF8 evidence")
	}
	d := json.NewDecoder(bytes.NewReader(raw))
	d.UseNumber()
	var read func(int) (any, error)
	read = func(depth int) (any, error) {
		if depth > 80 {
			return nil, errors.New("metadata depth")
		}
		t, e := d.Token()
		if e != nil {
			return nil, e
		}
		switch t {
		case json.Delim('{'):
			m := map[string]any{}
			for d.More() {
				k, e := d.Token()
				if e != nil {
					return nil, e
				}
				s := k.(string)
				if _, ok := m[s]; ok {
					return nil, errors.New("duplicate JSON key")
				}
				v, e := read(depth + 1)
				if e != nil {
					return nil, e
				}
				m[s] = v
			}
			_, e = d.Token()
			return m, e
		case json.Delim('['):
			a := []any{}
			for d.More() {
				v, e := read(depth + 1)
				if e != nil {
					return nil, e
				}
				a = append(a, v)
			}
			_, e = d.Token()
			return a, e
		default:
			return t, nil
		}
	}
	v, e := read(0)
	if e != nil {
		return nil, e
	}
	_, e = d.Token()
	if e != io.EOF {
		return nil, errors.New("trailing metadata")
	}
	return v, nil
}
func streamInt(v any) (int, bool) {
	n, ok := v.(json.Number)
	if !ok {
		return 0, false
	}
	i, e := n.Int64()
	return int(i), e == nil
}

func validateStreamChunk(raw []byte, j StreamJournal, roster []StreamRow) (StreamChunk, []byte, error) {
	var c StreamChunk
	if len(roster) != 373 {
		return c, nil, errors.New("complete immutable roster required")
	}
	normalized,_:=normalizeStreamTransport(raw)
	v, e := streamJSON(normalized)
	if e != nil {
		return c, nil, e
	}
	canonical, e := json.Marshal(v)
	if e != nil {
		return c, nil, e
	}
	decoder := json.NewDecoder(bytes.NewReader(canonical))
	decoder.UseNumber()
	if e = decoder.Decode(&c); e != nil {
		return c, nil, e
	}
	if c.Domain != "othergit373-chunk-v1" || c.Roster != streamRosterSHA || !regexp.MustCompile(`^[0-9a-f]{32}$`).MatchString(c.ID) || !regexp.MustCompile(`^\d{8}-\d{6}-todo-3778$`).MatchString(c.Job) || c.Todo != 3778 || c.First < 1 || len(c.Entries) == 0 || c.First+len(c.Entries)-1 > 373 {
		return c, nil, errors.New("chunk binding differs")
	}
	if c.Boundaries.Executed == nil || *c.Boundaries.Executed || c.Boundaries.Grants == nil || *c.Boundaries.Grants || c.Boundaries.Decisions == nil || *c.Boundaries.Decisions != 0 || c.Boundaries.Central == nil || *c.Boundaries.Central != 0 {
		return c, nil, errors.New("reading promoted to execution/grants/decisions")
	}
	for i, e := range c.Entries {
		ordinal, ok := streamInt(e["row"])
		if !ok || ordinal != c.First+i {
			return c, nil, errors.New("noncontiguous source rows")
		}
		src, ok := e["source"].([]any)
		if !ok || len(src) != 7 {
			return c, nil, errors.New("source tuple absent")
		}
		r := roster[ordinal-1]
		if r.Unit=="" || streamString(e["unit"])!=r.Unit { return c,nil,errors.New("source occurrence unit changed") }
		size, ok := streamInt(src[6])
		if !ok || streamString(src[0]) != r.ID || streamString(src[1]) != r.Entry.Repo || streamString(src[2]) != r.Entry.Commit || streamString(src[3]) != r.Entry.Path || streamString(src[4]) != r.Entry.SHA || streamString(src[5]) != r.Entry.Blob || size != r.Entry.Bytes {
			return c, nil, errors.New("primary identity differs from immutable roster")
		}
		for _, key := range []string{"use", "value", "reason", "positive", "negative", "candidate_disposition"} {
			if streamString(e[key]) == "" {
				return c, nil, errors.New("semantic assessment missing")
			}
		}
		read, ok := e["read"].([]any)
		if !ok || len(read) != 6 {
			return c, nil, errors.New("reading tuple absent")
		}
		state, ok := streamInt(read[0])
		mask, mok := streamInt(read[1])
		actor, aok := streamInt(read[2])
		last, lok := streamInt(read[5])
        if !ok || !lok { return c,nil,errors.New("reading state missing") }
        switch state {
        case 1:
            if read[1]!=nil || read[2]!=nil || read[3]!=false || read[4]!=false || last!=0 || streamString(e["block_reason"])=="" { return c,nil,errors.New("unread input gained a reader or completion") }
        case 6,7:
            if !mok || !aok || actor!=0 || mask<0 || last<1 || read[3]!=false || read[4]!=false || streamString(e["block_reason"])=="" { return c,nil,errors.New("partial/scan input promoted to full") }
            if _,ok:=e["unread_line_ranges"].([]any);!ok { return c,nil,errors.New("partial/scan unread ranges absent") }
        case 2,3:
            if !mok || !aok || actor!=0 || last<1 || read[4]!=true || read[3]!=(state==2) || state==2 && mask!=0 || state==3 && mask<1 { return c,nil,errors.New("raw/masked reading state differs") }
        default:
            return c,nil,errors.New("unknown primary reading state")
        }
		if _, ok := e["findings"].([]any); !ok {
			return c, nil, errors.New("findings absent")
		}
		if _, ok := e["required_changes"].([]any); !ok {
			return c, nil, errors.New("required changes absent")
		}
		if _, ok := e["notices"].([]any); !ok {
			return c, nil, errors.New("individual notices absent")
		}
		if _, ok := e["support"].([]any); !ok {
			return c, nil, errors.New("support states absent")
		}
	}
	return c, canonical, nil
}

	if c.ID != id || c.Job != r.Job || r.SHA != streamHash(canonical) {
		return errors.New("received evidence changed")
	}
	for _, done := range journal.Chunks {
		if done.Chunk == id {
			fmt.Printf("already acknowledged through%d\n", done.Last)
			return nil
		}
	}
	if c.First != journal.Through+1 {
		return errors.New("acceptance cursor differs")
	}
	journal.Previous = streamHash(old)
	journal.Generation++
	journal.Through = c.First + len(c.Entries) - 1
	journal.Chunks = append(journal.Chunks, StreamRecord{Chunk: id, Receipt: filepath.Base(file), SHA: r.SHA, Job: r.Job, First: c.First, Last: journal.Through, ParentSemanticRead: true})
	journal.Updated = time.Now().UTC().Format(time.RFC3339Nano)
	journal.ParentControl = nil
	if r.Normalization!=nil {
		journal.ParentControl=map[string]any{
			"domain":"parent3080-transport-unknown-value-accepted-v1",
			"consumer_job":os.Getenv("CC_JOB_ID"),"applies_only_to_reader_job":c.Job,
			"acknowledged_through":journal.Through,"unknown_value_count":r.Normalization.UnknownValues,
			"instruction":"親はこの範囲の意味評価を読了し受領した。原応答は不変で保存し、文字列外の[REDACTED]だけをunknownのJSON nullへ正規化した。伏せられた値・真偽・認証分類を復元/確定していない。旧syntax再送依頼は不要となった。同じ担当が次ordinalから続け、JSONをserialize/strict parseし、伏字はquoted stringまたは明示unknown nullで表す。原source/reader/mask数や権利・採否・中央を変更せず、過去本文を再表示しない。",
		}
	}
	if yieldReview!=0 {
		journal.ParentControl = map[string]any{
			"domain":"parent3080-resource-yield-v1",
			"consumer_job":os.Getenv("CC_JOB_ID"),
			"applies_only_to_reader_job":c.Job,

			"await_review_todo":yieldReview,
			"whole_research_complete":false,

	if e != nil {

	}
	if e = streamNew(filepath.Join(streamP, fmt.Sprintf("other373-journal-%04d-%s-074042.json", journal.Generation, c.ID)), updated); e != nil {
		return e
	}
	fmt.Printf("ACKNOWLEDGED semantic assessments through%d/373; grants0/decisions0/central0; supporting proof verification remains separate\n", journal.Through)
	return nil
}
