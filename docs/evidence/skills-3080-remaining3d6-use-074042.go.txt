func r6Hash(b []byte) string { h := sha256.Sum256(b); return hex.EncodeToString(h[:]) }
func r6Read(path string) ([]byte, error) {
	i, e := os.Lstat(path)
	if e != nil {
		return nil, e
	}
	if !i.Mode().IsRegular() || i.Size() > 2<<20 {
		return nil, errors.New("bounded regular file required")
	}
	w, e := syscall.UTF16PtrFromString(path)
	if e != nil {
		return nil, e
	}
	h, e := syscall.CreateFile(w, syscall.GENERIC_READ, syscall.FILE_SHARE_READ|syscall.FILE_SHARE_WRITE|syscall.FILE_SHARE_DELETE, nil, syscall.OPEN_EXISTING, syscall.FILE_FLAG_OPEN_REPARSE_POINT, 0)
	if e != nil {
		return nil, e
	}
	f := os.NewFile(uintptr(h), path)
	defer f.Close()
	st, e := f.Stat()
	var info syscall.ByHandleFileInformation
	if e != nil || !os.SameFile(i, st) || syscall.GetFileInformationByHandle(h, &info) != nil || info.FileAttributes&syscall.FILE_ATTRIBUTE_REPARSE_POINT != 0 || info.NumberOfLinks != 1 {
		return nil, errors.New("physical identity/link boundary differs")
	}
	b, e := io.ReadAll(io.LimitReader(f, 2<<20+1))
	if e != nil || len(b) > 2<<20 {
		return nil, errors.New("bounded read failed")
	}
	return b, nil
}

// Validate original token spelling before typed JSON decoding. No case alias or
// duplicate member is allowed; user-controlled map data is not present here.
func r6UniqueJSON(d *json.Decoder, depth int) error {
	if depth > 80 {
		return errors.New("receipt depth exceeds boundary")
	}
	tok, e := d.Token()
	if e != nil {
		return e
	}
	switch tok {
	case json.Delim('{'):
		seen := map[string]bool{}
		for d.More() {
			k, e := d.Token()
			if e != nil {
				return e
			}
			s, ok := k.(string)
			if !ok || seen[s] {
				return errors.New("duplicate receipt key")
			}
			seen[s] = true
			if e = r6UniqueJSON(d, depth+1); e != nil {
				return e
			}
		}
		_, e = d.Token()
		return e
	case json.Delim('['):
		for d.More() {
			if e = r6UniqueJSON(d, depth+1); e != nil {
				return e
			}
		}
		_, e = d.Token()
		return e
	}
	return nil
}
func r6DecodeExact(b []byte, allowed []string, dst any) error {
	d := json.NewDecoder(bytes.NewReader(b))
	if e := r6UniqueJSON(d, 0); e != nil {
		return e
	}
	if _, e := d.Token(); e != io.EOF {
		return errors.New("trailing JSON receipt")
	}
	var keys map[string]json.RawMessage
	if json.Unmarshal(b, &keys) != nil {
		return errors.New("receipt object required")
	}
	known := map[string]bool{}
	for _, k := range allowed {
		known[k] = true
	}
	for k := range keys {
		if !known[k] {
			return errors.New("unknown/case-aliased receipt field")
		}
	}
	// The exact allowed key set has already rejected unknown spellings. A
	// projection may intentionally omit allowed diagnostic fields below.
	return json.Unmarshal(b, dst)
}
func r6Command(b []byte, expected string) ([]byte, error) {
	if len(expected) != 64 || r6Hash(b) != expected {
		return nil, errors.New("authentic original wrapper differs")
	}
	var v struct {
		Exit  *int    `json:"exit"`
		Timed *bool   `json:"timed_out"`
		Out   *string `json:"stdout"`
		Err   *string `json:"stderr"`
	}
	if r6DecodeExact(b, []string{"exit", "timed_out", "stdout", "stderr"}, &v) != nil || v.Exit == nil || *v.Exit != 0 || v.Timed == nil || *v.Timed || v.Out == nil || *v.Out == "" || v.Err == nil || *v.Err != "" {
		return nil, errors.New("required process observation missing/failed")
	}
	return []byte(*v.Out), nil
}
