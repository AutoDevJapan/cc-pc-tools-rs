func useHash(b []byte) string { h:=sha256.Sum256(b);return hex.EncodeToString(h[:]) }

func useTimezones(t *testing.T) map[string]any {
	zb,e:=os.ReadFile(filepath.Join(runtime.GOROOT(),"lib","time","zoneinfo.zip"));if e!=nil{t.Fatal(e)}
	useNeed(t,useHash(zb)=="b2d18a7c8fa8142097a48c99609fb3c92db5ee98bc740294e57eab8ae9f94779","fixed tzdata changed; do not call it current by assumption")
	zr,e:=zip.NewReader(bytes.NewReader(zb),int64(len(zb)));if e!=nil{t.Fatal(e)}
	load:=func(name string)*time.Location{for _,f:=range zr.File{if f.Name!=name{continue};r,e:=f.Open();if e!=nil{t.Fatal(e)};b,e:=io.ReadAll(r);r.Close();if e!=nil{t.Fatal(e)};l,e:=time.LoadLocationFromTZData(name,b);if e!=nil{t.Fatal(e)};return l};t.Fatal("explicit zone missing");return nil}
	instant,e:=time.Parse(time.RFC3339,"2026-10-25T14:45:00Z");if e!=nil{t.Fatal(e)}
	want:=map[string]string{"Asia/Tokyo":"2026-10-25T23:45:00+09:00","Asia/Kathmandu":"2026-10-25T20:30:00+05:45","Australia/Adelaide":"2026-10-26T01:15:00+10:30"}
	converted:=map[string]string{};for name,expected:=range want{actual:=instant.In(load(name)).Format(time.RFC3339);useNeed(t,actual==expected,"explicit instant/date/offset conversion differs");converted[name]=actual}
	// Enumerate this bounded example's inverse mapping. Date/ParseInLocation on
	// an ambiguous local clock is deliberately not used to select an offset.
	inverse:=func(local string)[]string{anchor,e:=time.Parse("2006-01-02 15:04",local);if e!=nil{t.Fatal(e)};loc:=load("Australia/Lord_Howe");out:=[]string{};for at:=anchor.Add(-26*time.Hour);!at.After(anchor.Add(26*time.Hour));at=at.Add(time.Minute){if at.In(loc).Format("2006-01-02 15:04")==local{out=append(out,at.UTC().Format(time.RFC3339))}};return out}
	fold:=inverse("2026-04-05 01:45");gap:=inverse("2026-10-04 02:15")
	useNeed(t,reflect.DeepEqual(fold,[]string{"2026-04-04T14:45:00Z","2026-04-04T15:15:00Z"}),"half-hour fold must retain both UTC candidates")
	useNeed(t,len(gap)==0,"gap must not be silently normalized")
	return map[string]any{"input_utc":instant.Format(time.RFC3339),"converted":converted,"fold_local":"Australia/Lord_Howe 2026-04-05 01:45","fold_utc_candidates":fold,"gap_local":"Australia/Lord_Howe 2026-10-04 02:15","gap_utc_candidates":gap,"negative_manual_decision":"CSTの明日9時は地域と年月日が不足するため換算しない。これは実行した数値テストとは別の適用判断。","tzdata_source":"exact GOROOT/lib/time/zoneinfo.zip bytes loaded via LoadLocationFromTZData","tzdata_sha256":useHash(zb),"go_version":runtime.Version(),"cases":5,"limits":"Five bounded examples; inverse search is minute-resolution for the declared dates, not a universal resolver or current worldwide-rule guarantee. No calendar registration or messages."}
}

type useItem struct{ Instance string `json:"instance"`; Definition string `json:"definition"`; Owner string `json:"owner"` }
type useInventory struct{ Version int `json:"version"`; Bag useItem `json:"bag"`; Head useItem `json:"head"`; Applied map[string]int `json:"applied_ops"` }
func useClone(s useInventory)useInventory{m:=map[string]int{};for k,v:=range s.Applied{m[k]=v};s.Applied=m;return s}
func useEffect(s useInventory)int{if s.Head.Definition=="helmet"{return 10};return 0}
func useSwap(s useInventory,op string,expected int,saveFail bool)(useInventory,int,error){
	if version,ok:=s.Applied[op];ok{return s,version,nil}
	if s.Version!=expected{return s,0,errors.New("stale version")}
	if s.Bag.Instance==""||s.Head.Instance==""||s.Bag.Instance==s.Head.Instance||s.Bag.Owner!=s.Head.Owner||s.Bag.Definition!="helmet"{return s,0,errors.New("ownership or incompatible head slot")}
	next:=useClone(s);next.Bag,next.Head=next.Head,next.Bag;next.Version++;next.Applied[op]=next.Version
	if saveFail{return s,0,errors.New("save failed before commit")}
	return next,next.Version,nil
}
func useInventoryCases(t *testing.T)(useInventory,useInventory,map[string]any){
	before:=useInventory{1,useItem{"item-B","helmet","player-test"},useItem{"item-A","helmet","player-test"},map[string]int{}}
	after,version,e:=useSwap(before,"op-7",1,false);useNeed(t,e==nil&&version==2&&after.Bag.Instance=="item-A"&&after.Head.Instance=="item-B","full bag swap failed")
	ids:=func(s useInventory)[]string{v:=[]string{s.Bag.Instance,s.Head.Instance};sort.Strings(v);return v};useNeed(t,reflect.DeepEqual(ids(before),ids(after))&&useEffect(before)==10&&useEffect(after)==10,"instances lost or effect duplicated")
	replay,replayed,e:=useSwap(after,"op-7",1,false);useNeed(t,e==nil&&replayed==2&&reflect.DeepEqual(replay,after),"lost-response replay changed inventory/effects")
	bad:=useClone(after);bad.Bag.Definition="boots";unchanged,_,e:=useSwap(bad,"op-bad",2,false);useNeed(t,e!=nil&&reflect.DeepEqual(unchanged,bad),"incompatible slot changed either side")
	unchanged,_,e=useSwap(after,"op-save-fail",2,true);useNeed(t,e!=nil&&reflect.DeepEqual(unchanged,after)&&unchanged.Applied["op-save-fail"]==0,"save failure reported completion or changed state")
	unchanged,_,e=useSwap(after,"op-stale",1,false);useNeed(t,e!=nil&&reflect.DeepEqual(unchanged,after),"stale version changed state")
	return before,after,map[string]any{"bag_capacity":1,"instances_conserved":ids(after),"equipped_effect_before":useEffect(before),"equipped_effect_after":useEffect(after),"successful_operation":"op-7","replay_result_version":replayed,"incompatible_slot_preserves_both":true,"injected_save_failure_preserves_committed_state":true,"stale_version_rejected":true,"cases_before_disk_reload":5,"limits":"Reference state transitions and an injected pre-commit save failure, not a multi-process DB transaction or production game implementation."}
}

type useV3 struct{ X,Y,Z float64 }
type useWall struct{ X,Z,Radius,Low,High float64; Enabled bool }
func useCylinder(a,b useV3,w useWall)bool{
	if !w.Enabled{return false};lo,hi:=0.0,1.0;dx,dz:=b.X-a.X,b.Z-a.Z;ox,oz:=a.X-w.X,a.Z-w.Z;qa:=dx*dx+dz*dz;qc:=ox*ox+oz*oz-w.Radius*w.Radius
	if qa==0{if qc>0{return false}}else{qb:=2*(ox*dx+oz*dz);disc:=qb*qb-4*qa*qc;if disc<0{return false};r:=math.Sqrt(disc);lo=math.Max(lo,(-qb-r)/(2*qa));hi=math.Min(hi,(-qb+r)/(2*qa))}
	dy:=b.Y-a.Y;if dy==0{if a.Y<w.Low||a.Y>w.High{return false}}else{one,two:=(w.Low-a.Y)/dy,(w.High-a.Y)/dy;if one>two{one,two=two,one};lo=math.Max(lo,one);hi=math.Min(hi,two)}
	return lo<=hi
}
func useFog(t *testing.T)map[string]any{
	a,b:=useV3{0,0,0},useV3{10,10,0};wall:=useWall{5,0,2,4,6,true}
	useNeed(t,useCylinder(a,b,wall),"interior height overlap was missed by horizontal-boundary-only check")
	open:=wall;open.Enabled=false;useNeed(t,!useCylinder(a,b,open),"open gate still blocks")
	low:=wall;low.Low=0;low.High=2;useNeed(t,!useCylinder(a,b,low),"low box incorrectly blocks elevated sight")
	inside:=useV3{5,5,0};useNeed(t,useCylinder(inside,inside,wall),"declared closed-volume/zero-length inside case differs")
	distance:=math.Sqrt(200);within20:=distance<=20;within10:=distance<=10;useNeed(t,within20&&!within10,"range not separated from line of sight")
	// Selection consumes the same current CPU predicate; simulation/lifetime is
	// advanced separately and must remain active when the enemy is hidden.
	selected:="enemy-test";ticks:=0;alive:=true;for _,w:=range []useWall{open,wall}{visible:=within20&&!useCylinder(a,b,w);ticks++;if !visible{selected=""}}
	useNeed(t,selected==""&&ticks==2&&alive,"visibility stopped simulation/lifetime or retained an invalid target")
	return map[string]any{"crossing_horizontal_interval":"[0.3,0.7]","crossing_height_interval":"[0.4,0.6]","closed_wall_blocks":true,"open_gate_clear_same_state":true,"low_box_clear":true,"zero_length_inside_closed_volume_blocks":true,"range20_contains_range10_excludes":true,"invalid_target_cleared":true,"hidden_enemy_simulation_ticks":ticks,"hidden_enemy_alive":alive,"cases":6,"fixture_boundary_rule":"Closed solid volume includes its boundary; zero-length inside blocks in this declared fixture.","renderer_executed":false,"actual_game_executed":false,"ui_changed":false,"performance_measured":false,"limits":"Pure CPU geometry and state-consumer example only; no terrain reconstruction, GPU, browser, production AI or rendering acceptance."}
